import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { randomUUID } from "node:crypto";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

export async function POST(req: NextRequest) {
  try {
    const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
    const anonKey =
      process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY ||
      process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
    const serviceKey = process.env.SUPABASE_SERVICE_ROLE_KEY;
    const paystackKey = process.env.PAYSTACK_SECRET_KEY;

    if (!url || !anonKey || !serviceKey) {
      return NextResponse.json(
        { error: "Supabase server configuration is incomplete." },
        { status: 503 },
      );
    }
    if (!paystackKey) {
      return NextResponse.json(
        { error: "Paystack is not configured." },
        { status: 503 },
      );
    }

    const authorization = req.headers.get("authorization") || "";
    const match = authorization.match(/^Bearer\s+(.+)$/i);
    if (!match) {
      return NextResponse.json({ error: "Sign in is required." }, { status: 401 });
    }

    const authClient = createClient(url, anonKey, {
      auth: { persistSession: false, autoRefreshToken: false },
    });
    const { data: authData, error: authError } =
      await authClient.auth.getUser(match[1]);

    if (authError || !authData.user || !authData.user.email) {
      return NextResponse.json({ error: "Your session is invalid. Sign in again." }, { status: 401 });
    }

    const body = await req.json();
    const rawAmount = String(body?.amount ?? "").trim();

    if (!/^\d{1,7}(\.\d{1,2})?$/.test(rawAmount)) {
      return NextResponse.json({ error: "Enter a valid amount in GHS." }, { status: 400 });
    }

    const amount = Number(rawAmount);
    const pesewas = Math.round(amount * 100);

    if (!Number.isSafeInteger(pesewas) || pesewas < 100 || pesewas > 100000000) {
      return NextResponse.json({ error: "Amount must be between GHS 1.00 and GHS 1,000,000.00." }, { status: 400 });
    }

    const admin = createClient(url, serviceKey, {
      auth: { persistSession: false, autoRefreshToken: false },
    });

    const reference = `SB-${randomUUID()}`;
    const { error: insertError } = await admin
      .from("paystack_transactions")
      .insert({
        user_id: authData.user.id,
        email: authData.user.email,
        reference,
        amount: (pesewas / 100).toFixed(2),
        currency: "GHS",
        status: "initialized",
        metadata: { platform: "skillbridge" },
      });

    if (insertError) {
      console.error("Unable to create payment record:", insertError.message);
      return NextResponse.json({ error: "Could not create the payment record." }, { status: 500 });
    }

    const siteUrl = (process.env.NEXT_PUBLIC_SITE_URL || process.env.NEXT_PUBLIC_URL || req.nextUrl.origin).replace(/\/+$/, "");
    const gatewayResponse = await fetch("https://api.paystack.co/transaction/initialize", {
      method: "POST",
      headers: {
        Authorization: `Bearer ${paystackKey}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        email: authData.user.email,
        amount: pesewas,
        currency: "GHS",
        reference,
        callback_url: `${siteUrl}/dashboard?payment=return&reference=${encodeURIComponent(reference)}`,
        metadata: { user_id: authData.user.id, platform: "skillbridge" },
      }),
      cache: "no-store",
    });

    const gateway = await gatewayResponse.json();

    if (!gatewayResponse.ok || !gateway.status || !gateway.data?.authorization_url) {
      await admin.from("paystack_transactions")
        .update({ status: "failed", updated_at: new Date().toISOString() })
        .eq("reference", reference)
        .eq("status", "initialized");

      console.error("Paystack initialization failed:", gateway.message || gateway.status);
      return NextResponse.json({ error: "Paystack could not initialize this payment." }, { status: 502 });
    }

    return NextResponse.json({
      authorization_url: gateway.data.authorization_url,
      reference,
    });
  } catch (error) {
    console.error("Payment initialization error:", error);
    return NextResponse.json({ error: "Unable to start payment." }, { status: 500 });
  }
}

import { createHmac, timingSafeEqual } from "node:crypto";
import { createClient } from "@supabase/supabase-js";
import { NextResponse } from "next/server";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

export async function POST(request: Request) {
  const secret = process.env.PAYSTACK_SECRET_KEY;
  const signature = request.headers.get("x-paystack-signature");

  if (!secret) {
    return NextResponse.json({ error: "Payment service is not configured." }, { status: 503 });
  }
  if (!signature || !/^[a-f0-9]{128}$/i.test(signature)) {
    return NextResponse.json({ error: "Invalid Paystack signature." }, { status: 401 });
  }

  const rawBody = await request.text();
  const expected = createHmac("sha512", secret).update(rawBody).digest();
  const supplied = Buffer.from(signature, "hex");

  if (supplied.length !== expected.length || !timingSafeEqual(supplied, expected)) {
    return NextResponse.json({ error: "Invalid Paystack signature." }, { status: 401 });
  }

  let event: { event?: string; data?: { reference?: string } };
  try {
    event = JSON.parse(rawBody);
  } catch {
    return NextResponse.json({ error: "Invalid webhook payload." }, { status: 400 });
  }

  if (event.event !== "charge.success") {
    return NextResponse.json({ received: true, ignored: true });
  }

  const reference = event.data?.reference;
  if (!reference) {
    return NextResponse.json({ error: "Payment reference is missing." }, { status: 400 });
  }

  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const serviceKey = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !serviceKey) {
    console.error("Missing server-side Supabase configuration for webhook.");
    return NextResponse.json({ error: "Payment processing is not configured." }, { status: 503 });
  }

  const admin = createClient(url, serviceKey, {
    auth: { persistSession: false, autoRefreshToken: false },
  });

  try {
    const { data: payment, error: lookupError } = await admin
      .from("paystack_transactions")
      .select("user_id,reference,amount,currency,status")
      .eq("reference", reference)
      .maybeSingle();

    if (lookupError) throw lookupError;
    if (!payment || !payment.user_id) {
      console.error("Unknown Paystack reference:", reference);
      return NextResponse.json({ error: "Payment reference not found." }, { status: 404 });
    }

    if (payment.status === "success") {
      return NextResponse.json({ received: true, processed: true, already_processed: true });
    }

    if (payment.status !== "initialized") {
      return NextResponse.json({ received: true, ignored: true });
    }

    const verifyResponse = await fetch(
      `https://api.paystack.co/transaction/verify/${encodeURIComponent(reference)}`,
      {
        headers: { Authorization: `Bearer ${secret}` },
        cache: "no-store",
      },
    );

    const verification = await verifyResponse.json();
    const verified = verification?.data;

    if (!verifyResponse.ok || !verification?.status || !verified) {
      console.error("Paystack verification request failed for reference:", reference);
      return NextResponse.json({ error: "Could not verify payment." }, { status: 502 });
    }

    const expectedPesewas = Math.round(Number(payment.amount) * 100);
    if (
      verified.status !== "success" ||
      !Number.isSafeInteger(verified.id) ||
      verified.id <= 0 ||
      verified.reference !== reference ||
      verified.currency !== "GHS" ||
      !Number.isSafeInteger(expectedPesewas) ||
      verified.amount !== expectedPesewas
    ) {
      console.error("Paystack verification mismatch for reference:", reference);
      return NextResponse.json({ error: "Payment verification did not match the saved transaction." }, { status: 400 });
    }

    const { data: creditResult, error: creditError } = await admin.rpc(
      "credit_paystack_wallet",
      {
        p_reference: reference,
        p_paystack_transaction_id: verified.id,
        p_amount: Number(payment.amount),
        p_currency: verified.currency,
        p_user_id: payment.user_id,
      },
    );

    if (creditError) {
      console.error("Wallet credit RPC failed:", creditError.message);
      return NextResponse.json({ error: "Wallet credit failed; Paystack may retry this webhook." }, { status: 500 });
    }

    const creditPayload = Array.isArray(creditResult)
      ? creditResult[0]
      : creditResult;

    if (
      !creditPayload ||
      typeof creditPayload !== "object" ||
      creditPayload.success !== true
    ) {
      console.error("Paystack wallet credit RPC did not explicitly confirm success.");
      return NextResponse.json(
        { error: "Wallet credit was not confirmed. Paystack may retry this webhook." },
        { status: 500 },
      );
    }

    return NextResponse.json({ received: true, processed: true, result: creditResult });
  } catch (error) {
    console.error("Paystack webhook processing error:", error);
    return NextResponse.json({ error: "Payment processing failed; retry is possible." }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

export async function GET(request: NextRequest) {
  try {
    const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
    const anonKey =
      process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY ||
      process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY;

    if (!supabaseUrl || !anonKey) {
      return NextResponse.json(
        { error: "Wallet service is not configured." },
        { status: 503 },
      );
    }

    const authorization = request.headers.get("authorization");
    const token = authorization?.match(/^Bearer\s+(.+)$/i)?.[1];

    if (!token) {
      return NextResponse.json(
        { error: "Please sign in to view your wallet." },
        { status: 401 },
      );
    }

    const supabase = createClient(supabaseUrl, anonKey, {
      auth: {
        persistSession: false,
        autoRefreshToken: false,
      },
      global: {
        headers: {
          Authorization: `Bearer ${token}`,
        },
      },
    });

    const { data: authData, error: authError } =
      await supabase.auth.getUser(token);

    if (authError || !authData.user) {
      return NextResponse.json(
        { error: "Your session is invalid or expired." },
        { status: 401 },
      );
    }

    const { data: wallet, error: walletError } = await supabase
      .from("platform_wallets")
      .select("available_balance, total_earnings")
      .eq("user_id", authData.user.id)
      .maybeSingle();

    if (walletError) {
      console.error("Wallet balance lookup failed:", walletError.message);
      return NextResponse.json(
        { error: "Wallet balance is temporarily unavailable." },
        { status: 503 },
      );
    }

    const balance = wallet ? Number(wallet.available_balance) : 0;

    if (!Number.isFinite(balance) || balance < 0) {
      return NextResponse.json(
        { error: "Wallet balance is invalid." },
        { status: 500 },
      );
    }

    return NextResponse.json({
      balance,
      currency: "GHS",
    });
  } catch (error) {
    console.error("Wallet balance API failed:", error);
    return NextResponse.json(
      { error: "Wallet balance is temporarily unavailable." },
      { status: 500 },
    );
  }
}

CREATE OR REPLACE FUNCTION public.credit_paystack_wallet(
  p_reference text,
  p_paystack_transaction_id bigint,
  p_amount numeric,
  p_currency text,
  p_user_id uuid
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, pg_temp
AS $$
DECLARE
  payment_row public.paystack_transactions%ROWTYPE;
BEGIN
  IF p_reference IS NULL OR p_reference = ''
     OR p_paystack_transaction_id IS NULL OR p_paystack_transaction_id <= 0
     OR p_amount IS NULL OR p_amount <= 0
     OR p_currency <> 'GHS' OR p_user_id IS NULL THEN
    RETURN jsonb_build_object('success', false, 'error', 'Invalid payment parameters.');
  END IF;

  SELECT * INTO payment_row
  FROM public.paystack_transactions
  WHERE reference = p_reference
  FOR UPDATE;

  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'error', 'Payment reference not found.');
  END IF;

  IF payment_row.user_id <> p_user_id
     OR payment_row.currency <> p_currency
     OR payment_row.amount <> p_amount THEN
    RETURN jsonb_build_object('success', false, 'error', 'Payment details do not match.');
  END IF;

  IF payment_row.status = 'success' THEN
    IF EXISTS (
      SELECT 1 FROM public.paystack_wallet_ledger
      WHERE reference = p_reference
        AND user_id = p_user_id
        AND amount = p_amount
        AND paystack_transaction_id = p_paystack_transaction_id
    ) THEN
      RETURN jsonb_build_object('success', true, 'already_processed', true);
    END IF;

    RETURN jsonb_build_object('success', false, 'error', 'Payment marked successful without matching ledger entry.');
  END IF;

  IF payment_row.status <> 'initialized' THEN
    RETURN jsonb_build_object('success', false, 'error', 'Payment is not awaiting credit.');
  END IF;

  IF EXISTS (
    SELECT 1 FROM public.paystack_wallet_ledger
    WHERE reference = p_reference
  ) THEN
    RETURN jsonb_build_object('success', false, 'error', 'Ledger reference already exists.');
  END IF;

  INSERT INTO public.paystack_wallet_ledger
    (reference, paystack_transaction_id, user_id, amount, currency)
  VALUES
    (p_reference, p_paystack_transaction_id, p_user_id, p_amount, p_currency);

  INSERT INTO public.platform_wallets
    (user_id, available_balance, total_earnings, updated_at)
  VALUES
    (p_user_id, p_amount, p_amount, now())
  ON CONFLICT (user_id) DO UPDATE
  SET available_balance = public.platform_wallets.available_balance + EXCLUDED.available_balance,
      total_earnings = public.platform_wallets.total_earnings + EXCLUDED.total_earnings,
      updated_at = now();

  UPDATE public.paystack_transactions
  SET status = 'success',
      paystack_transaction_id = p_paystack_transaction_id,
      updated_at = now()
  WHERE reference = p_reference;

  RETURN jsonb_build_object('success', true, 'already_processed', false);
END;
$$;
